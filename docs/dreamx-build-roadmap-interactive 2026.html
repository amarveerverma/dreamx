1-Click Pre-Flight Health Check Script
scripts/preflight_check.py (Automatic Connectivity Test)
Run this script with Antigravity IDE before starting Phase 0. It tests connections to Supabase, OpenAI, Razorpay, Resend, and Sentry in under 5 seconds! 
import os, sys, requests

def check_all():
    print("🚀 Running DreamX v1 Pre-Flight Verification (5 Services)...\n")
    results = {}

    # 1. Supabase Ping
    sb_url = os.getenv("NEXT_PUBLIC_SUPABASE_URL") or os.getenv("SUPABASE_URL")
    sb_key = os.getenv("NEXT_PUBLIC_SUPABASE_ANON_KEY")
    if sb_url and sb_key:
        try:
            r = requests.get(f"{sb_url}/rest/v1/", headers={"apikey": sb_key}, timeout=5)
            results["Supabase DB & Auth"] = "✅ CONNECTED (200 OK)" if r.status_code in [200, 401] else f"⚠️ {r.status_code}"
        except Exception as e:
            results["Supabase DB & Auth"] = f"❌ FAILED: {str(e)}"
    else:
        results["Supabase DB & Auth"] = "⚠️ MISSING ENV KEYS"

    # 2. OpenAI GPT-4o Ping
    openai_key = os.getenv("OPENAI_API_KEY")
    if openai_key:
        try:
            r = requests.get("https://api.openai.com/v1/models", headers={"Authorization": f"Bearer {openai_key}"}, timeout=5)
            results["OpenAI GPT-4o"] = "✅ CONNECTED (Models Loaded)" if r.status_code == 200 else f"⚠️ {r.status_code}"
        except Exception as e:
            results["OpenAI GPT-4o"] = f"❌ FAILED: {str(e)}"
    else:
        results["OpenAI GPT-4o"] = "⚠️ MISSING ENV KEYS"

    # 3. Razorpay Gateway Ping
    rzp_id = os.getenv("RAZORPAY_KEY_ID")
    rzp_secret = os.getenv("RAZORPAY_KEY_SECRET")
    if rzp_id and rzp_secret:
        try:
            r = requests.get("https://api.razorpay.com/v1/payments?count=1", auth=(rzp_id, rzp_secret), timeout=5)
            results["Razorpay Gateway"] = "✅ CONNECTED" if r.status_code == 200 else f"⚠️ {r.status_code}"
        except Exception as e:
            results["Razorpay Gateway"] = f"❌ FAILED: {str(e)}"
    else:
        results["Razorpay Gateway"] = "⚠️ MISSING ENV KEYS"

    # 4. Resend Email Ping
    resend_key = os.getenv("RESEND_API_KEY")
    if resend_key:
        try:
            r = requests.get("https://api.resend.com/domains", headers={"Authorization": f"Bearer {resend_key}"}, timeout=5)
            results["Resend Email"] = "✅ CONNECTED" if r.status_code == 200 else f"⚠️ {r.status_code}"
        except Exception as e:
            results["Resend Email"] = f"❌ FAILED: {str(e)}"
    else:
        results["Resend Email"] = "⚠️ MISSING ENV KEYS"

    # 5. Sentry DSN Check
    sentry_dsn = os.getenv("SENTRY_DSN")
    if sentry_dsn and sentry_dsn.startswith("https://"):
        results["Sentry DSN"] = "✅ CONFIGURED"
    else:
        results["Sentry DSN"] = "⚠️ MISSING OR INVALID DSN"

    for svc, status in results.items():
        print(f"[{status.split()[0]}] {svc}: {' '.join(status.split()[1:])}")

    all_good = all("✅" in s for s in results.values())
    if all_good:
        print("\n🎉 ALL 5 v1 SERVICES PRE-CONNECTED & VERIFIED! READY FOR PHASE 0.")
    else:
        print("\n⚠️ Please resolve missing or failing keys before starting Phase 0.")

if __name__ == "__main__":
    check_all()
