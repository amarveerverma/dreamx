02

Event Router & State Machine

PostgreSQL ACID
event: "payment.captured" • "order.paid" Transition: Confirmed
  • Atomic DB Update: service_bookings.status ← 'confirmed' with row lock FOR UPDATE.
  • Escrow Hold Initialization: Sets payout_status = 'held'. Service bookings lock for 3 days; Structured Programs lock for 14 days post-milestone.
  • Split Ledger Split: 80% Mentor Credited in Escrow Ledger / 20% DreamX Platform Commission retained + 18% GST audit record.
  • Event Bus Broadcast: Emits PaymentSucceededEvent & BookingConfirmedEvent.
event: "payment.failed" Transition: Cancelled
  • Releases the temporary 15-minute slot lock reservation in mentor_schedules.
  • Updates service_bookings.status ← 'cancelled'.
  • Emits PaymentFailedEvent to alert student via push notification to retry via alternate gateway route.
event: "refund.processed" Transition: Refunded
  • Releases escrow hold lock in payout_transactions to payout_status = 'refunded'.
  • Revokes active Google Meet conference permissions via calendar integration.

Production Implementation

src/app/api/webhooks/razorpay/route.ts HMAC SHA256 • Supabase Transact 
import { NextRequest, NextResponse } from 'next/server';
import crypto from 'node:crypto';
import { supabaseAdmin } from '@/lib/supabase/admin';

export async function POST(req: NextRequest) {
  const rawBody = await req.text();
  const signature = req.headers.get('x-razorpay-signature');
  const secret = process.env.RAZORPAY_WEBHOOK_SECRET!;

  // 1. Strict HMAC Verification
  const expectedSig = crypto
    .createHmac('sha256', secret)
    .update(rawBody)
    .digest('hex');

  if (!signature || !crypto.timingSafeEqual(Buffer.from(signature), Buffer.from(expectedSig))) {
    return NextResponse.json({ error: 'Invalid HMAC signature' }, { status: 401 });
  }

  const payload = JSON.parse(rawBody);
  const eventId = req.headers.get('x-razorpay-event-id') || payload.payload.payment.entity.id;

  // 2. Idempotency Lock Guard
  const { data: existing } = await supabaseAdmin
    .from('audit.webhook_events')
    .insert({ event_id: eventId, event_name: payload.event, raw_payload: payload })
    .select('id')
    .single();

  if (!existing) {
    return NextResponse.json({ status: 'ignored_idempotent_duplicate' }, { status: 200 });
  }

  // 3. Process Captured Transaction
  if (payload.event === 'payment.captured' || payload.event === 'order.paid') {
    const payment = payload.payload.payment.entity;
    const bookingId = payment.notes.booking_id;
    const isProgram = payment.notes.is_program === 'true';
    const holdInterval = isProgram ? '14 days' : '3 days';

    // Atomic DB State Machine + Escrow Hold
    await supabaseAdmin.rpc('process_successful_payment_escrow', {
      p_booking_id: bookingId,
      p_payment_id: payment.id,
      p_order_id: payment.order_id,
      p_amount_paise: payment.amount,
      p_hold_interval: holdInterval
    });

    // 4. Trigger Async Workers (Meet URL & Calendar)
    await inngest.send({
      name: 'dreamx/booking.confirmed',
      data: { bookingId, paymentId: payment.id }
    });
  }

  return NextResponse.json({ status: 'processed', timestamp: Date.now() }, { status: 200 });
}
